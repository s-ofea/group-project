hd_mortalitybyca<- read.csv("data/hd_mortalitybyca.csv")
hd_mortalitybyca %>%
  filter(
  Year != "2024p", 
  Ca == "S92000003",
  Diagnosis == "Coronary Heart Disease",
  AgeGroup %in% c("0-44 years", "45-64 years", "65-74 years", "75plus years"),
  Sex != "All") %>%
    mutate(Year = as.numeric(Year)) %>%
  
  
  ggplot(aes(x = Year, y = CrudeRate, colour = Sex)) + 
  geom_point(alpha = 0.6) + 
  geom_smooth(method = "lm", se = FALSE) + 
  facet_wrap(~AgeGroup, scales = "free_y", axes = "all") +
  
  theme_minimal(base_size = 13) +
  theme(
    plot.title = element_text(face = "bold", size = 14, margin = margin(b = 10)),
    plot.margin = margin(15, 20, 15, 15),
    axis.title = element_text(face = "bold"),
    plot.subtitle = element_text(size = 10)
  ) +
  scale_x_continuous(breaks = seq(2015, 2023, by = 2)) +
  
  labs(
    title = "Coronary Heart Disease Mortality Rates in Scotland (2015-2023)",
    subtitle = "By Age Group",
    x = "Year (2015-2023)",
    y = "Crude Mortality Rate (per 100 000 population)"
  )
